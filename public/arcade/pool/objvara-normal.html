// Función creadora de la vara 3D modular
function createPoolCue(THREE, config = {}) {
  const group = new THREE.Group();

  const shaftMat = new THREE.MeshStandardMaterial({
    color: config.shaftColor || '#d9a95f',
    roughness: config.roughness ?? 0.35,
    metalness: config.metalness ?? 0.05
  });
  const shaft = new THREE.Mesh(new THREE.CylinderGeometry(2.2, 4.6, 132, 28), shaftMat);
  shaft.rotation.z = -Math.PI / 2;
  shaft.position.x = 12;
  group.add(shaft);

  const buttMat = new THREE.MeshStandardMaterial({
    color: config.buttColor || '#141418',
    roughness: config.roughness ?? 0.35,
    metalness: config.metalness ?? 0.25
  });
  const butt = new THREE.Mesh(new THREE.CylinderGeometry(4.6, 5.6, 54, 28), buttMat);
  butt.rotation.z = -Math.PI / 2;
  butt.position.x = -81;
  group.add(butt);

  const ringMat = new THREE.MeshStandardMaterial({
    color: config.ringsColor || '#fbbf24',
    metalness: 0.9, roughness: 0.18
  });
  const ring1 = new THREE.Mesh(new THREE.CylinderGeometry(4.75, 4.75, 3.8, 28), ringMat);
  ring1.rotation.z = -Math.PI / 2; ring1.position.x = -54; group.add(ring1);

  const tipMat = new THREE.MeshStandardMaterial({ color: config.tipColor || '#4aa3df' });
  const tip = new THREE.Mesh(new THREE.CylinderGeometry(2.0, 2.18, 5.5, 28), tipMat);
  tip.rotation.z = -Math.PI / 2; tip.position.x = 86.8; group.add(tip);

  return group;
}
